# 名贵商品监控 - 脚本效果
# 检查单个名贵商品：已被解锁过+日志未激活+存在符合条件的潜在生产商 -> 重新添加日志
# 参数: $GOODS$ $UNLOCK_VAR$ $JE$
pgs_check_single_good = {
	if = {
		limit = {
			any_company = { has_variable = $UNLOCK_VAR$ }
			NOT = { has_journal_entry = $JE$ }
			any_company = {
				can_potentially_produce_prestige_goods = $GOODS$
				is_producing_prestige_goods = no
				company_is_prosperous = yes
				NOT = { has_variable = $UNLOCK_VAR$ }
			}
		}
		add_journal_entry = { type = $JE$ }
		change_variable = { name = pgs_readded_count add = 1 }
	}
}
