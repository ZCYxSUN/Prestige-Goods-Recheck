# 名贵商品监控 - 脚本效果
# 检查单个名贵商品：已被解锁过+日志未激活+存在符合条件的潜在生产商 -> 重新添加日志
# 参数: $GOODS$ $UNLOCK_VAR$ $JE$
# ===== 兼容接口(契约) =====
# 本模组的"检测层"(监控面板/原版名贵商品日志)依赖三个通用触发:
#   can_potentially_produce_prestige_goods / is_producing_prestige_goods / company_is_prosperous
#   以及公司上的解锁变量 <prestige_good_<id>_var>。
# 因此【新增公司】无需任何改动:只要其名贵商品 JE 使用上述原版触发、并在公司上设置解锁变量,
# 就会被本模组自动识别。
# 【新增名贵商品】需要一步:在下方新增一行 pgs_check_single_good=<GOODS/UNLOCK_VAR/JE> ...
# 建议新增商品放入"扩展钩子 pgs_extra_prestige_good_checks"(见底部说明),避免改本文件。
pgs_check_single_good = {
	if = {
		limit = {
			any_company = { has_variable = $UNLOCK_VAR$ }
			NOT = { has_journal_entry = $JE$ }
			any_company = {
				can_potentially_produce_prestige_goods = $GOODS$
				is_producing_prestige_goods = no
				company_is_prosperous = yes
				NOT = { has_variable = $UNLOCK_VAR$ }
			}
		}
		add_journal_entry = { type = $JE$ }
		change_variable = { name = pgs_readded_count add = 1 }
	}
}
# 对全部16种名贵商品执行检查(国家作用域)
