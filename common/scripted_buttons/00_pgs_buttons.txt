# 名贵商品监控 - 脚本按钮
# 立即检查按钮：立即执行全商品检查
pgs_check_now_button = {
	name = pgs_check_now_button
	desc = pgs_check_now_button_desc

	visible = { always = yes }
	possible = { always = yes }

	effect = {
		pgs_run_all_checks = yes
		custom_tooltip = pgs_check_executed_tt
	}
}

# 关闭AI自动检查按钮：设置全局变量，使所有AI国家跳过定期检查(玩家自身检查不受影响)
pgs_disable_ai_check_button = {
	name = pgs_disable_ai_check_button
	desc = pgs_disable_ai_check_button_desc

	visible = { always = yes }
	possible = { always = yes }

	# 已关闭时高亮，便于实时反映当前状态(无需重新读取游戏)
	selected = {
		has_global_variable = pgs_ai_check_disabled
	}

	effect = {
		set_global_variable = { name = pgs_ai_check_disabled value = 1 }
	}
}

# 启用AI自动检查按钮：移除全局变量，恢复AI国家的定期检查
pgs_enable_ai_check_button = {
	name = pgs_enable_ai_check_button
	desc = pgs_enable_ai_check_button_desc

	visible = { always = yes }
	possible = { always = yes }

	# 已启用(即未禁用)时高亮
	selected = {
		NOT = { has_global_variable = pgs_ai_check_disabled }
	}

	effect = {
		remove_global_variable = pgs_ai_check_disabled
	}
}
